== Honor `#address-cells` and `#size-cells` with `reg` in node children ==
node {
  #address-cells = <3>;
  #size-cells = <2>;

  child {
    reg = <0x1 0x2 0x3 0x4 0x5 0x6 0x7 0x8 0x9 0x10>;
  };
};

[expect]
node {
  #address-cells = <3>;
  #size-cells = <2>;

  child {
    reg = <0x1 0x2 0x3 0x4 0x5
           0x6 0x7 0x8 0x9 0x10>;
  };
};

== Ensure the cell values only apply locally to reg values ==
node-1 {
  reg = <0x0 0x1 0x2 0x3 0x4 0x5 0x6 0x7 0x8 0x9 0xa 0xb 0xc 0xd 0xe 0xf>;
  node-2 {
    #address-cells = <3>;
    #size-cells = <4>;
    reg = <0x0 0x1 0x2 0x3 0x4 0x5 0x6 0x7 0x8 0x9 0xa 0xb 0xc 0xd 0xe 0xf>;

    node-3 {
      #address-cells = <2>;
      #size-cells = <2>;
      reg = <0x0 0x1 0x2 0x3 0x4 0x5 0x6 0x7 0x8 0x9 0xa 0xb 0xc 0xd 0xe 0xf>;
    };

    // node-4 inherits 3 + 4 from node-2
    node-4 {
      reg = <0x0 0x1 0x2 0x3 0x4 0x5 0x6 0x7 0x8 0x9 0xa 0xb 0xc 0xd 0xe 0xf>;
    };
  };

  // node-5 should fall back to the 2 + 1 default
  node-5 {
    reg = <0x0 0x1 0x2 0x3 0x4 0x5 0x6 0x7 0x8 0x9 0xa 0xb 0xc 0xd 0xe 0xf>;
  };
};
